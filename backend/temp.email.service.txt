Yes. For the mail layer, I would use **four pieces**:

```text
config/mail.js
services/email.service.js
controllers/auth.controller.js     ← add verification methods
controllers/project.controller.js  ← completion notification trigger/queue later
```

For now, here is the complete mail implementation for **verification email + project completion email**, with secure token generation, hashed token storage, expiration, HTML + plaintext email, SMTP verification, and safe error handling.

Nodemailer supports reusable SMTP transporters, `transporter.verify()` for checking connectivity/authentication, and separate `text`/`html` message bodies. ([Nodemailer][1])

# 1. Install Nodemailer

```bash
npm install nodemailer
```

# 2. `.env`

Add:

```env
# ============================================================
# EMAIL / SMTP
# ============================================================

SMTP_HOST=smtp.example.com
SMTP_PORT=587
SMTP_SECURE=false
SMTP_USER=your-smtp-username
SMTP_PASS=your-smtp-password

MAIL_FROM="Subtitle App <no-reply@example.com>"

# Frontend
FRONTEND_URL=https://your-frontend.com

# Verification link validity
EMAIL_VERIFICATION_EXPIRES_HOURS=24

# Optional
MAIL_REPLY_TO=support@example.com
```

For SMTP port `587`, `SMTP_SECURE=false` is normally appropriate and STARTTLS is used; port `465` is typically configured with `secure=true`. ([Nodemailer][1])

Never commit `.env`.

# 3. `config/mail.js`

This file should **only configure the mail transport**.

Do not put application email logic here.

```js
"use strict";

const nodemailer = require("nodemailer");


// ============================================================
// ENVIRONMENT
// ============================================================

const SMTP_HOST =
  process.env.SMTP_HOST;

const SMTP_PORT = Number(
  process.env.SMTP_PORT || 587
);

const SMTP_SECURE =
  String(
    process.env.SMTP_SECURE || "false"
  ).toLowerCase() === "true";

const SMTP_USER =
  process.env.SMTP_USER;

const SMTP_PASS =
  process.env.SMTP_PASS;

const MAIL_FROM =
  process.env.MAIL_FROM;

const MAIL_REPLY_TO =
  process.env.MAIL_REPLY_TO ||
  undefined;


// ============================================================
// VALIDATION
// ============================================================

const requiredEnvironmentVariables = [
  "SMTP_HOST",
  "SMTP_USER",
  "SMTP_PASS",
  "MAIL_FROM",
];

for (
  const variable
  of requiredEnvironmentVariables
) {
  if (
    !process.env[variable]
  ) {
    throw new Error(
      `${variable} is required for email service`
    );
  }
}


// ============================================================
// SMTP TRANSPORT
// ============================================================
//
// One transporter is created and reused.
//
// Do NOT create a new transporter for every email.
// Nodemailer supports connection pooling/reuse.
// ============================================================

const transporter =
  nodemailer.createTransport({
    host: SMTP_HOST,

    port: SMTP_PORT,

    secure: SMTP_SECURE,

    auth: {
      user: SMTP_USER,
      pass: SMTP_PASS,
    },

    // --------------------------------------------------------
    // Connection pooling
    // --------------------------------------------------------

    pool: true,

    maxConnections: Number(
      process.env.SMTP_MAX_CONNECTIONS ||
        5
    ),

    maxMessages: Number(
      process.env.SMTP_MAX_MESSAGES ||
        100
    ),

    // --------------------------------------------------------
    // Timeouts
    // --------------------------------------------------------

    connectionTimeout:
      Number(
        process.env.SMTP_CONNECTION_TIMEOUT ||
          10000
      ),

    greetingTimeout:
      Number(
        process.env.SMTP_GREETING_TIMEOUT ||
          10000
      ),

    socketTimeout:
      Number(
        process.env.SMTP_SOCKET_TIMEOUT ||
          30000
      ),

    // --------------------------------------------------------
    // Message defaults
    // --------------------------------------------------------

    defaults: {
      from: MAIL_FROM,

      ...(MAIL_REPLY_TO
        ? {
            replyTo:
              MAIL_REPLY_TO,
          }
        : {}),
    },
  });


// ============================================================
// VERIFY SMTP
// ============================================================

const verifyMailConnection =
  async () => {
    try {
      await transporter.verify();

      console.log(
        "[MAIL] SMTP connection verified successfully"
      );

      return true;
    } catch (error) {
      console.error(
        "[MAIL] SMTP verification failed:",
        error.message
      );

      return false;
    }
  };


// ============================================================
// EXPORT
// ============================================================

module.exports = {
  transporter,
  verifyMailConnection,
};
```

The reusable transporter and `verify()` pattern are directly supported by Nodemailer. `verify()` checks DNS resolution, connection, TLS upgrade where applicable, and authentication without sending an email. ([Nodemailer][1])

const sendEmail = async ({
  to,
  subject,
  text,
  html,
  headers = {},
}) => {
  if (!to) {
    throw new Error(
      "Email recipient is required"
    );
  }

  if (!subject) {
    throw new Error(
      "Email subject is required"
    );
  }

  if (!text) {
    throw new Error(
      "Plain text email body is required"
    );
  }

  if (!html) {
    throw new Error(
      "HTML email body is required"
    );
  }

  try {
    const info =
      await transporter.sendMail({
        to,
        subject,
        text,
        html,

        headers: {
          ...headers,

          "X-Application":
            "Subtitle-App",
        },
      });

    console.log(
      `[MAIL] Sent "${subject}" to ${to} | messageId=${info.messageId}`
    );

    return {
      success: true,
      messageId:
        info.messageId,
      accepted:
        info.accepted,
      rejected:
        info.rejected,
      response:
        info.response,
    };
  } catch (error) {
    console.error(
      `[MAIL] Failed "${subject}" to ${to}:`,
      error.message
    );

    throw error;
  }
};

# 7. Important email-send failure handling

Don't necessarily delete the user if SMTP temporarily fails.

For example:

```text
User created ✅
Verification data saved ✅
SMTP failed ❌
```

The account can remain:

```text
isEmailVerified = false
```

and the user can use:

```text
POST /api/auth/resend-verification
```

to try again.

This is much better than making SMTP availability part of database account creation.

# 8. `verifyEmail` controller

Add this to `auth.controller.js`.

```js
const crypto = require("crypto");

const User = require("../models/User");

const {
  hashVerificationToken,
} = require("../services/email.service");


// ============================================================
// VERIFY EMAIL
// ============================================================

const verifyEmail = async (
  req,
  res
) => {
  try {
    const {
      token,
    } = req.body;

    if (
      !token ||
      typeof token !==
        "string"
    ) {
      return res.status(400).json({
        success: false,
        message:
          "Verification token is required",
      });
    }

    const normalizedToken =
      token.trim();

    if (
      normalizedToken.length <
        20 ||
      normalizedToken.length >
        200
    ) {
      return res.status(400).json({
        success: false,
        message:
          "Invalid verification token",
      });
    }

    // --------------------------------------------------------
    // Hash supplied token
    // --------------------------------------------------------

    const tokenHash =
      hashVerificationToken(
        normalizedToken
      );

    // --------------------------------------------------------
    // Find matching user
    //
    // tokenHash is select:false,
    // so explicitly select it.
    // --------------------------------------------------------

    const user =
      await User.findOne({
        "emailVerification.tokenHash":
          tokenHash,
      }).select(
        "+emailVerification.tokenHash +emailVerification.expiresAt"
      );

    if (!user) {
      return res.status(400).json({
        success: false,
        message:
          "Invalid or expired verification link",
      });
    }

    // --------------------------------------------------------
    // Already verified
    // --------------------------------------------------------

    if (
      user.isEmailVerified
    ) {
      return res.status(200).json({
        success: true,
        message:
          "Email address is already verified",
      });
    }

    // --------------------------------------------------------
    // Check expiration
    // --------------------------------------------------------

    if (
      !user.emailVerification
        ?.expiresAt ||
      user.emailVerification
        .expiresAt.getTime() <=
        Date.now()
    ) {
      // Invalidate expired token.
      user.emailVerification.tokenHash =
        null;

      user.emailVerification.expiresAt =
        null;

      await user.save();

      return res.status(410).json({
        success: false,
        message:
          "Verification link has expired. Please request a new one.",
      });
    }

    // --------------------------------------------------------
    // Verify
    // --------------------------------------------------------

    user.isEmailVerified =
      true;

    user.emailVerification
      .verifiedAt =
      new Date();

    // Single-use token
    user.emailVerification
      .tokenHash =
      null;

    user.emailVerification
      .expiresAt =
      null;

    await user.save();

    return res.status(200).json({
      success: true,
      message:
        "Email verified successfully",
      data: {
        isEmailVerified: true,
      },
    });
  } catch (error) {
    console.error(
      "verifyEmail error:",
      error
    );

    return res.status(500).json({
      success: false,
      message:
        "Unable to verify email",
    });
  }
};
```

# 9. Resend verification controller

This is equally important.

```js
const {
  createEmailVerificationToken,
  sendVerificationEmail,
} = require("../services/email.service");


// ============================================================
// RESEND VERIFICATION
// ============================================================

const resendVerificationEmail =
  async (
    req,
    res
  ) => {
    try {
      const email =
        String(
          req.body.email ||
            ""
        )
          .trim()
          .toLowerCase();

      if (!email) {
        return res.status(400).json({
          success: false,
          message:
            "Email is required",
        });
      }

      // ------------------------------------------------------
      // Find user
      // ------------------------------------------------------

      const user =
        await User.findOne({
          email,
        }).select(
          "+emailVerification.tokenHash +emailVerification.expiresAt"
        );

      // ------------------------------------------------------
      // Do not expose whether an email exists.
      //
      // This prevents account enumeration.
      // ------------------------------------------------------

      if (!user) {
        return res.status(200).json({
          success: true,
          message:
            "If an account exists for this email, a verification email will be sent.",
        });
      }

      // ------------------------------------------------------
      // Already verified
      // ------------------------------------------------------

      if (
        user.isEmailVerified
      ) {
        return res.status(200).json({
          success: true,
          message:
            "Email address is already verified",
        });
      }

      // ------------------------------------------------------
      // Account disabled
      // ------------------------------------------------------

      if (
        !user.isActive
      ) {
        return res.status(200).json({
          success: true,
          message:
            "If an account exists for this email, a verification email will be sent.",
        });
      }

      // ------------------------------------------------------
      // Rate limit
      // ------------------------------------------------------

      const now =
        Date.now();

      const lastSent =
        user.emailVerification
          ?.lastSentAt
          ? user.emailVerification
              .lastSentAt
              .getTime()
          : 0;

      const resendCooldown =
        60 * 1000;

      if (
        lastSent &&
        now - lastSent <
          resendCooldown
      ) {
        return res.status(429).json({
          success: false,
          message:
            "Please wait before requesting another verification email.",
          retryAfterSeconds:
            Math.ceil(
              (
                resendCooldown -
                (
                  now -
                  lastSent
                )
              ) / 1000
            ),
        });
      }

      // ------------------------------------------------------
      // Generate new token
      // ------------------------------------------------------

      const verification =
        createEmailVerificationToken();

      user.emailVerification = {
        tokenHash:
          verification.tokenHash,

        expiresAt:
          verification.expiresAt,

        verifiedAt:
          null,

        lastSentAt:
          new Date(),

        sendCount:
          (
            user
              .emailVerification
              ?.sendCount ||
            0
          ) + 1,
      };

      await user.save();

      // ------------------------------------------------------
      // Send
      // ------------------------------------------------------

      try {
        await sendVerificationEmail({
          name:
            user.name,

          email:
            user.email,

          token:
            verification.token,
        });
      } catch (error) {
        console.error(
          "Resend verification email failed:",
          error
        );

        return res.status(202).json({
          success: true,
          message:
            "Verification request accepted. Please try again shortly if the email does not arrive.",
        });
      }

      return res.status(200).json({
        success: true,
        message:
          "Verification email sent",
      });
    } catch (error) {
      console.error(
        "resendVerificationEmail error:",
        error
      );

      return res.status(500).json({
        success: false,
        message:
          "Unable to resend verification email",
      });
    }
  };
```
